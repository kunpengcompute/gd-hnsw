function(ub_instrument target)
  if(UB_TEST_COVERAGE)
    target_compile_options(${target} PRIVATE --coverage -O0 -g -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE --coverage)
  endif()
endfunction()

function(ub_add_test target source unit_filter functional_filter)
  add_executable(${target} "${CMAKE_CURRENT_SOURCE_DIR}/${source}" "${CMAKE_CURRENT_SOURCE_DIR}/test_main.cpp")
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(${target} PRIVATE ub_test_support GTest::gtest)
  target_compile_definitions(${target} PRIVATE HNSW_URMA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  ub_instrument(${target})
  if(NOT "${unit_filter}" STREQUAL "")
    add_test(NAME "unit.${target}" COMMAND ${target} "--gtest_filter=${unit_filter}" "--gtest_output=xml:${UB_REPORT_DIR}/gtest/unit.${target}.xml")
    set_tests_properties("unit.${target}" PROPERTIES LABELS unit TIMEOUT 180 RUN_SERIAL TRUE)
  endif()
  if(NOT "${functional_filter}" STREQUAL "")
    add_test(NAME "functional.${target}" COMMAND ${target} "--gtest_filter=${functional_filter}" "--gtest_output=xml:${UB_REPORT_DIR}/gtest/functional.${target}.xml")
    set_tests_properties("functional.${target}" PROPERTIES LABELS functional TIMEOUT 180 RUN_SERIAL TRUE)
  endif()
endfunction()
